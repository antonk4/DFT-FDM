#version 330 core

layout (location=0) in vec4 vertex_position;
layout (location=1) in uint occlusion;

uniform mat4 projectionMatrix;
uniform mat4 viewMatrix;

out vec4 fragment_color;

void main() {
  gl_Position = vec4(projectionMatrix * viewMatrix * vec4(vertex_position.xyz, 1.0));
  fragment_color = vec4(min((occlusion+4.0)/8.0, 4.0*occlusion/4.0*(0.5+vertex_position.w)/ gl_Position.z / gl_Position.z), 0.0, min((occlusion+4.0)/8.0, 4.0*occlusion/4.0*(0.5-vertex_position.w)/ gl_Position.z / gl_Position.z), 1.0);
}

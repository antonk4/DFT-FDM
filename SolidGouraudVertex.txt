#version 330 core

layout (location=0) in vec4 vertex_position;

uniform mat4 projectionMatrix;
uniform mat4 viewMatrix;

out vec4 fragment_color;

void main() {
  gl_Position = vec4(projectionMatrix * viewMatrix * vec4(vertex_position.xyz, 1.0));
  fragment_color = 2*vec4(0.5+vertex_position.w, 0.0, 0.5-vertex_position.w, gl_Position.z * gl_Position.z)/ gl_Position.z / gl_Position.z;
}
